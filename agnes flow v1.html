#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ image_to_video_youtube.py — Générateur complet d'images, vidéos et métadonnées YouTube. Pipeline : 1. Choix du style + mood → génération d'image (Replicate SDXL, fallback DALL·E 3) 2. Animation de l'image (Replicate SVD, fallback effet parallaxe local FFmpeg) 3. Génération des métadonnées YouTube (OpenAI GPT, fallback templates) Exécution : python image_to_video_youtube.py """ import os import sys import time import json import logging import subprocess import tempfile import shutil from pathlib import Path from typing import Dict, Optional, Tuple, Any # ─── Dépendances externes ──────────────────────────────────────────────── try: import requests from dotenv import load_dotenv except ImportError as e: print(f"[ERREUR] Dépendance manquante : {e}") print("Installez les dépendances avec : pip install -r requirements.txt") sys.exit(1) # ─── Configuration ─────────────────────────────────────────────────────── load_dotenv() OUTPUT_DIR = Path("output") OUTPUT_DIR.mkdir(exist_ok=True) logging.basicConfig( level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s", datefmt="%H:%M:%S", ) log = logging.getLogger(__name__) # Clés API (facultatives — des fallbacks existent) REPLICATE_API_TOKEN = os.getenv("REPLICATE_API_TOKEN", "") OPENAI_API_KEY = os.getenv("OPENAI_API_KEY", "") # ─── Catalogue de styles et moods ──────────────────────────────────────── STYLES = { "ghibli": { "label": "Studio Ghibli", "prompt": "Studio Ghibli anime style, hand-painted watercolor backgrounds, " "soft natural light, lush nature, gentle atmosphere, cel-shaded characters", }, "pixar": { "label": "Pixar 3D", "prompt": "Pixar 3D animation style, cinematic CGI render, soft global illumination, " "expressive characters, vibrant colors, family-friendly", }, "anime": { "label": "Anime 2020s", "prompt": "Modern 2020s anime style, crisp digital line art, vibrant colors, " "cinematic lighting, detailed backgrounds, expressive eyes", }, "watercolor": { "label": "Aquarelle", "prompt": "Watercolor painting style, soft washes, pastel palette, " "delicate brush strokes, paper texture, dreamy atmosphere", }, "noir": { "label": "Film Noir", "prompt": "Film noir black and white, high contrast, dramatic shadows, " "moody atmosphere, classic 1940s aesthetic, cigarette smoke", }, } MOODS = { "cozy": { "label": "Cozy", "prompt": "cozy warm atmosphere, soft golden lighting, intimate feeling, " "comfortable and welcoming, warm color palette", "animation": {"motion_strength": 30, "fps": 8}, }, "chill": { "label": "Chill", "prompt": "relaxing chill mood, soothing colors, calm environment, " "peaceful and quiet, smooth ambience", "animation": {"motion_strength": 40, "fps": 8}, }, "relaxing": { "label": "Relaxing", "prompt": "deeply relaxing scene, meditative atmosphere, soft lighting, " "tranquil and serene, no tension", "animation": {"motion_strength": 25, "fps": 6}, }, "night": { "label": "Night", "prompt": "night atmosphere, deep blues, moonlit, glowing lights, " "mysterious and cinematic, stars or city lights", "animation": {"motion_strength": 60, "fps": 10}, }, "apaisant": { "label": "Apaisant", "prompt": "soothing peaceful scene, harmonious composition, " "gentle natural light, calm and reassuring", "animation": {"motion_strength": 30, "fps": 8}, }, } DEFAULT_BASE_PROMPT = ( "a peaceful landscape with rolling hills and soft clouds, " "highly detailed, beautiful composition, cinematic quality" ) # ─── Utilitaires ───────────────────────────────────────────────────────── def _download(url: str, dest: Path) -> Path: """Télécharge une URL vers un fichier local.""" log.info(f"Téléchargement : {url[:80]}…") r = requests.get(url, stream=True, timeout=180) r.raise_for_status() with open(dest, "wb") as f: for chunk in r.iter_content(chunk_size=8192): f.write(chunk) log.info(f"Fichier sauvegardé : {dest} ({dest.stat().st_size // 1024} Ko)") return dest def _check_ffmpeg() -> bool: """Vérifie que ffmpeg est disponible dans le PATH.""" return shutil.which("ffmpeg") is not None def _poll_replicate_prediction(prediction: Dict[str, Any], max_wait: int = 300) -> Dict[str, Any]: """Poll une prédiction Replicate jusqu'à completion ou échec.""" headers = {"Authorization": f"Token {REPLICATE_API_TOKEN}"} poll_url = prediction.get("urls", {}).get("get") if not poll_url: raise RuntimeError("URL de polling Replicate manquante") deadline = time.time() + max_wait while time.time() < deadline: r = requests.get(poll_url, headers=headers, timeout=30) r.raise_for_status() pred = r.json() status = pred.get("status") if status == "succeeded": log.info("Prédiction Replicate terminée.") return pred if status in ("failed", "canceled"): raise RuntimeError(f"Prédiction échouée : {pred.get('error')}") log.info(f"En attente… statut={status}") time.sleep(3) raise TimeoutError("Délai d'attente Replicate dépassé") # ─── Étape 1 — Génération d'image ──────────────────────────────────────── def generate_image(style_key: str, mood_key: str, base_prompt: str) -> Path: """ Génère une image en combinant style + mood + prompt de base. Essaie Replicate (SDXL), puis DALL·E 3 en fallback. """ style = STYLES.get(style_key, STYLES["ghibli"]) mood = MOODS.get(mood_key, MOODS["cozy"]) full_prompt = ( f"{base_prompt}, {style['prompt']}, {mood['prompt']}, " "masterpiece, best quality, highly detailed, 4k, cinematic" ) log.info(f"Prompt complet : {full_prompt}") timestamp = int(time.time()) out_path = OUTPUT_DIR / f"image_{style_key}_{mood_key}_{timestamp}.png" # --- Tentative 1 : Replicate SDXL --- if REPLICATE_API_TOKEN: try: log.info("Génération via Replicate (SDXL)…") headers = { "Authorization": f"Token {REPLICATE_API_TOKEN}", "Content-Type": "application/json", } payload = { "version": "stability-ai/sdxl:39ed52f2a78e934b3ba6e2a89f5b1c712de7dfea535525255b1aa35c5565e08b", "input": { "prompt": full_prompt, "width": 1024, "height": 1024, "num_outputs": 1, "num_inference_steps": 30, "guidance_scale": 7.5, }, } r = requests.post( "https://api.replicate.com/v1/predictions", headers=headers, json=payload, timeout=30, ) r.raise_for_status() pred = _poll_replicate_prediction(r.json()) output = pred.get("output") if isinstance(output, list): output = output[0] return _download(output, out_path) except Exception as e: log.warning(f"Replicate a échoué : {e} — fallback DALL·E 3") # --- Tentative 2 : OpenAI DALL·E 3 --- if OPENAI_API_KEY: try: log.info("Génération via OpenAI DALL·E 3…") r = requests.post( "https://api.openai.com/v1/images/generations", headers={ "Authorization": f"Bearer {OPENAI_API_KEY}", "Content-Type": "application/json", }, json={ "model": "dall-e-3", "prompt": full_prompt[:4000], "n": 1, "size": "1024x1024", "quality": "hd", }, timeout=180, ) r.raise_for_status() url = r.json()["data"][0]["url"] return _download(url, out_path) except Exception as e: log.error(f"DALL·E 3 a échoué : {e}") # --- Fallback : image placeholder générée localement --- log.warning("Aucune API d'image disponible — génération d'un placeholder local.") try: from PIL import Image, ImageDraw except ImportError: raise RuntimeError( "Impossible de générer une image : aucune API disponible et Pillow absent." ) img = Image.new("RGB", (1024, 1024), color=(30, 30, 60)) draw = ImageDraw.Draw(img) draw.text((40, 40), f"PLACEHOLDER\n{style['label']} — {mood['label']}", fill=(200, 200, 255)) draw.text((40, 120), base_prompt[:80], fill=(180, 180, 220)) img.save(out_path) log.info(f"Placeholder sauvegardé : {out_path}") return out_path # ─── Étape 2 — Animation de l'image ────────────────────────────────────── def _animate_with_replicate(image_path: Path, mood_key: str) -> Optional[Path]: """Anime via Stable Video Diffusion sur Replicate.""" if not REPLICATE_API_TOKEN: return None mood = MOODS.get(mood_key, MOODS["cozy"]) motion = mood["animation"]["motion_strength"] try: log.info("Animation via Replicate (Stable Video Diffusion)…") # Upload de l'image d'abord with open(image_path, "rb") as f: files = {"content": f} r = requests.post( "https://api.replicate.com/v1/files", headers={"Authorization": f"Token {REPLICATE_API_TOKEN}"}, files=files, timeout=60, ) r.raise_for_status() image_url = r.json()["urls"]["get"] headers = { "Authorization": f"Token {REPLICATE_API_TOKEN}", "Content-Type": "application/json", } payload = { "version": "stability-ai/stable-video-diffusion:3f0457e4619daac51203dedb472816fd4af51f3149fa7a9e0b5ffcf1b8172438", "input": { "input_image": image_url, "video_length": "25_frames_with_svd_xt", "sizing_strategy": "maintain_aspect_ratio", "frames_per_second": mood["animation"]["fps"], "motion_bucket_id": motion, "cond_aug": 0.02, }, } r = requests.post( "https://api.replicate.com/v1/predictions", headers=headers, json=payload, timeout=30, ) r.raise_for_status() pred = _poll_replicate_prediction(r.json(), max_wait=600) video_url = pred.get("output") if isinstance(video_url, list): video_url = video_url[0] timestamp = int(time.time()) out_path = OUTPUT_DIR / f"video_{mood_key}_{timestamp}.mp4" return _download(video_url, out_path) except Exception as e: log.warning(f"Animation Replicate échouée : {e}") return None def _animate_with_parallax(image_path: Path, mood_key: str) -> Optional[Path]: """Effet parallaxe local via FFmpeg (fallback).""" if not _check_ffmpeg(): log.warning("FFmpeg introuvable — impossible d'animer localement.") return None mood = MOODS.get(mood_key, MOODS["cozy"]) duration = 5 fps = mood["animation"]["fps"] timestamp = int(time.time()) out_path = OUTPUT_DIR / f"video_{mood_key}_{timestamp}.mp4" # Zoom lent + panoramique léger filter_complex = ( f"scale=1024:1024," f"zoompan=z='min(zoom+0.0015,1.5)':d={duration * fps}:" f"x='iw/2-(iw/zoom/2)':y='ih/2-(ih/zoom/2)':" f"s=1024x1024:fps={fps}" ) cmd = [ "ffmpeg", "-y", "-loop", "1", "-i", str(image_path), "-vf", filter_complex, "-c:v", "libx264", "-t", str(duration), "-pix_fmt", "yuv420p", "-preset", "medium", str(out_path), ] log.info(f"Effet parallaxe local FFmpeg → {out_path}") try: result = subprocess.run(cmd, capture_output=True, text=True, timeout=120) if result.returncode != 0: log.error(f"FFmpeg erreur : {result.stderr[:300]}") return None log.info(f"Vidéo parallaxe créée : {out_path}") return out_path except Exception as e: log.error(f"Parallaxe FFmpeg échouée : {e}") return None def animate_image(image_path: Path, mood_key: str) -> Path: """Anime l'image — essaie Replicate SVD puis fallback parallaxe FFmpeg.""" log.info("Étape 2 — Animation de l'image…") video = _animate_with_replicate(image_path, mood_key) if video: return video log.warning("Replicate indisponible ou échec — fallback parallaxe local.") video = _animate_with_parallax(image_path, mood_key) if video: return video # Dernier recours : copier l'image en tant que "vidéo" (dégradé) log.error("Aucune méthode d'animation n'a fonctionné — copie de l'image en MP4.") fallback = OUTPUT_DIR / f"video_fallback_{int(time.time())}.mp4" if _check_ffmpeg(): subprocess.run( ["ffmpeg", "-y", "-loop", "1", "-i", str(image_path), "-c:v", "libx264", "-t", "3", "-pix_fmt", "yuv420p", str(fallback)], capture_output=True, timeout=60, ) return fallback # ─── Étape 3 — Métadonnées YouTube ─────────────────────────────────────── def _youtube_metadata_template(style_key: str, mood_key: str, base_prompt: str) -> Dict[str, str]: """Templates intelligents de secours pour les métadonnées YouTube.""" style = STYLES.get(style_key, STYLES["ghibli"]) mood = MOODS.get(mood_key, MOODS["cozy"]) title = ( f"✨ {style['label']} · {mood['label']} Vibes — Relaxing Ambience 🎨" ) description = ( f"🌿 Plongez dans une ambiance {mood['label'].lower()} avec ce visuel {style['label']}.\n\n" f"🎨 Style : {style['label']}\n" f"🌙 Ambiance : {mood['label']}\n" f"🖼️ Scène : {base_prompt[:120]}\n\n" "Idéal pour se détendre, étudier, dormir, méditer ou simplement s'évader.\n\n" "💡 Abonnez-vous pour plus d'ambiances apaisantes !\n\n" f"#ambiance #relaxing #chill #cozy #{style_key.lower()} #{mood_key.lower()} " "#loop #sleep #study #meditation #cinematic" ) tags = [ style["label"], mood["label"], "relaxing", "ambience", "study music", "sleep", "loop", "chill", "cozy", style_key, mood_key, "cinematic", "aesthetic", ] return {"title": title, "description": description, "tags": ", ".join(tags)} def generate_youtube_metadata(style_key: str, mood_key: str, base_prompt: str) -> Dict[str, str]: """ Génère un titre + description optimisés YouTube via OpenAI GPT. Fallback : templates intelligents si l'API est indisponible. """ log.info("Étape 3 — Génération des métadonnées YouTube…") style = STYLES.get(style_key, STYLES["ghibli"]) mood = MOODS.get(mood_key, MOODS["cozy"]) if not OPENAI_API_KEY: log.warning("OPENAI_API_KEY manquante — utilisation des templates intelligents.") return _youtube_metadata_template(style_key, mood_key, base_prompt) try: prompt = ( "Tu es un expert en SEO YouTube spécialisé dans les ambiances relaxantes. " "Génère un titre accrocheur (max 70 caractères) et une description engageante " "(200-400 mots, avec émojis, hashtags pertinents et structure claire) pour " "une vidéo relaxante.\n\n" f"Style : {style['label']} ({style['prompt']})\n" f"Ambiance : {mood['label']} ({mood['prompt']})\n" f"Scène : {base_prompt}\n\n" "Réponds au format JSON strict : " '{"title": "...", "description": "...", "tags": "tag1, tag2, ..."}' ) r = requests.post( "https://api.openai.com/v1/chat/completions", headers={ "Authorization": f"Bearer {OPENAI_API_KEY}", "Content-Type": "application/json", }, json={ "model": "gpt-4o-mini", "messages": [ {"role": "system", "content": "Tu es un expert SEO YouTube. Réponds uniquement en JSON."}, {"role": "user", "content": prompt}, ], "temperature": 0.8, "max_tokens": 800, "response_format": {"type": "json_object"}, }, timeout=60, ) r.raise_for_status() data = json.loads(r.json()["choices"][0]["message"]["content"]) log.info("Métadonnées YouTube générées via OpenAI.") return { "title": data.get("title", "")[:100], "description": data.get("description", ""), "tags": data.get("tags", ""), } except Exception as e: log.warning(f"OpenAI a échoué : {e} — fallback templates.") return _youtube_metadata_template(style_key, mood_key, base_prompt) # ─── Menu interactif ───────────────────────────────────────────────────── def _choose(prompt: str, options: Dict[str, Dict]) -> str: """Affiche un menu et retourne la clé choisie.""" keys = list(options.keys()) print("\n" + prompt) for i, k in enumerate(keys, 1): print(f" {i}. {options[k]['label']}") while True: try: choice = int(input("Votre choix : ").strip()) if 1 <= choice <= len(keys): return keys[choice - 1] except (ValueError, EOFError): pass print("Choix invalide, réessayez.") # ─── Main ──────────────────────────────────────────────────────────────── def main() -> int: """Point d'entrée principal.""" print("=" * 60) print(" 🎬 Générateur d'images, vidéos et métadonnées YouTube") print("=" * 60) if not REPLICATE_API_TOKEN and not OPENAI_API_KEY: log.warning( "Aucune clé API détectée. Le script utilisera des fallbacks locaux " "(placeholder image + parallaxe FFmpeg + templates SEO)." ) if not _check_ffmpeg(): log.warning("FFmpeg n'est pas installé — l'animation locale sera indisponible.") # 1. Choix utilisateur style_key = _choose("Choisissez un style visuel :", STYLES) mood_key = _choose("Choisissez une ambiance :", MOODS) base_prompt = input( f"\nDescription de base (Entrée pour : \"{DEFAULT_BASE_PROMPT[:50]}…\") : " ).strip() or DEFAULT_BASE_PROMPT # 2. Pipeline try: image_path = generate_image(style_key, mood_key, base_prompt) video_path = animate_image(image_path, mood_key) metadata = generate_youtube_metadata(style_key, mood_key, base_prompt) except Exception as e: log.error(f"Erreur fatale : {e}") return 1 # 3. Résumé final print("\n" + "=" * 60) print(" ✅ GÉNÉRATION TERMINÉE") print("=" * 60) print(f"🖼️ Image : {image_path}") print(f"🎞️ Vidéo : {video_path}") print(f"\n📺 Titre YouTube :\n {metadata['title']}") print(f"\n📝 Description YouTube :\n{metadata['description']}") print(f"\n🏷️ Tags :\n {metadata['tags']}") # 4. Sauvegarde JSON des métadonnées meta_file = OUTPUT_DIR / f"metadata_{style_key}_{mood_key}.json" with open(meta_file, "w", encoding="utf-8") as f: json.dump( { "style": STYLES[style_key]["label"], "mood": MOODS[mood_key]["label"], "base_prompt": base_prompt, "image_path": str(image_path), "video_path": str(video_path), **metadata, }, f, ensure_ascii=False, indent=2, ) print(f"\n💾 Métadonnées sauvegardées : {meta_file}") return 0 if __name__ == "__main__": sys.exit(main())
